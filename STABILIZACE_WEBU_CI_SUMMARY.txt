app.js - Fetch rate limiting:

PŘIDÁNO:
function checkFetchRateLimit() {
  const now = Date.now();
  fetchHistory = fetchHistory.filter(t => now - t < FETCH_RATE_LIMIT.windowMs);
  if (fetchHistory.length >= FETCH_RATE_LIMIT.count) {
    const oldest = fetchHistory[0];
    const waitMs = FETCH_RATE_LIMIT.windowMs - (now - oldest);
    if (waitMs > 0) {
      console.warn(`[infoUzel] Fetch rate limit: ${fetchHistory.length} fetches in ${FETCH_RATE_LIMIT.windowMs}ms`);
      return waitMs;
    }
  }
  fetchHistory.push(now);
  return 0;
}

app.js - localStorage fallback:

PŘIDÁNO:
function getCachedData(key) {
  try {
    const cached = localStorage.getItem(`iu:cache:${key}`);
    if (cached) {
      const parsed = JSON.parse(cached);
      const age = Date.now() - (parsed.timestamp || 0);
      if (age < 3600000) { // 1 hodina
        return parsed.data;
      }
    }
  } catch (e) {}
  return null;
}

function setCachedData(key, data) {
  try {
    localStorage.setItem(`iu:cache:${key}`, JSON.stringify({
      timestamp: Date.now(),
      data: data
    }));
  } catch (e) {}
}
