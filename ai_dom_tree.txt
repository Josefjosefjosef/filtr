console.log("AI panel parent chain:", (function(){
  const el = document.querySelector('#iu-aiPanel') || document.querySelector('[data-section="ai"]');
  if(!el) return null;
  const chain = [];
  let n = el;
  for(let i=0;i<8 && n;i++){
    chain.push(n.tagName.toLowerCase() + (n.id ? '#'+n.id : '') + (n.className ? '.'+String(n.className).trim().replace(/\s+/g,'.') : ''));
    n = n.parentElement;
  }
  return chain;
})());
