=== Produkce – computed pozadí (spusť v Console při otevřeném AI panelu) ===

const el = document.querySelector('#iu-aiPanel') || document.querySelector('[data-section="ai"]');
if(el){
  const cs = getComputedStyle(el);
  console.log("AI BG:", cs.backgroundColor, cs.backgroundImage);
  const p = el.parentElement;
  if(p){
    const ps = getComputedStyle(p);
    console.log("PARENT:", p, "BG:", ps.backgroundColor, ps.backgroundImage);
  }
}
